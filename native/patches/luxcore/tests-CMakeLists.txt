project(turninator_luxcore_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(loader_tests loader-tests.cpp "${SDK_SOURCE}/deps/cuew/src/cuew.cpp")
target_include_directories(loader_tests PRIVATE "${SDK_SOURCE}/deps/cuew/include")
target_compile_definitions(loader_tests PRIVATE NOMINMAX _ITERATOR_DEBUG_LEVEL=0)
add_executable(cache_tests cache-tests.cpp)

target_link_libraries(cache_tests PRIVATE bcrypt)

target_compile_definitions(direct_consumer PRIVATE NOMINMAX WIN32 BOOST_ALL_NO_LIB _ITERATOR_DEBUG_LEVEL=0)

target_link_libraries(direct_consumer PRIVATE "${SDK_LIBRARY}")

target_link_libraries(render_probe PRIVATE "${SDK_LIBRARY}" bcrypt)
