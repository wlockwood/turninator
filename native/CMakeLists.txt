cmake_minimum_required(VERSION 3.22)
project(turninator_renderers LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native ABI1 is qualified only for Windows x64 MSVC")
endif()
if(NOT MSVC)
  message(FATAL_ERROR "Use MSVC with the pinned dynamic Release CRT")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk-lock.json" tn_sdk_lock)
foreach(part header license)
  string(JSON relative GET "${tn_sdk_lock}" common_parser ${part}_path)
  string(JSON expected GET "${tn_sdk_lock}" common_parser ${part}_sha256)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Vendored schema1 JSON parser ${part} differs from sdk-lock.json")
  endif()
endforeach()
add_library(tn_native_core STATIC common/facade.cpp common/validation.cpp)
target_include_directories(tn_native_core PUBLIC common include vendor)
target_compile_definitions(tn_native_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tn_native_core PRIVATE /W4 /WX /EHsc)

# Both production bridges use precisely this facade and their own backend.cpp.
function(tn_add_bridge name backend_source)
  add_library(${name} SHARED ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/common/exports.cpp ${backend_source})
  target_link_libraries(${name} PRIVATE tn_native_core)
  target_compile_definitions(${name} PRIVATE TN_BUILDING_BRIDGE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${name} PRIVATE /W4 /WX /EHsc)
endfunction()
option(TN_BUILD_FIXTURE "Build CPU-only test DLLs (never production renderers)" OFF)
set(TN_BACKEND "" CACHE STRING "Production backend: filament or luxcore")
set(TN_SDK_ROOT "" CACHE PATH "Verified pinned SDK input root")
if(TN_BUILD_FIXTURE)
  enable_testing()
  if(NOT TN_BACKEND STREQUAL "")
    message(FATAL_ERROR "Fixture and production backend builds must be separate")
  endif()
  add_subdirectory(fixture)
elseif(TN_BACKEND STREQUAL "filament" OR TN_BACKEND STREQUAL "luxcore")
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${TN_BACKEND}/CMakeLists.txt")
    message(FATAL_ERROR "Production ${TN_BACKEND} bridge is not implemented; fixture is not a substitute")
  endif()
  if(NOT IS_DIRECTORY "${TN_SDK_ROOT}")
    message(FATAL_ERROR "Provide a verified pinned TN_SDK_ROOT")
  endif()
  add_subdirectory(${TN_BACKEND})
else()
  message(FATAL_ERROR "Select a production TN_BACKEND or explicitly enable TN_BUILD_FIXTURE")
endif()
