  foreach(index RANGE 0 ${tn_artifacts_last})
    string(JSON path GET "${tn_sdk_lock}" inputs ${index} path)
    if(path STREQUAL "filament/${relative}")
      string(JSON expected GET "${tn_sdk_lock}" inputs ${index} sha256)
      file(SHA256 "${TN_SDK_ROOT}/${relative}" actual)
      if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Filament pinned input differs: ${relative}")
      endif()
      set(found ON)
    endif()
  endforeach()
  if(NOT found)
    message(FATAL_ERROR "Filament input is absent from sdk-lock: ${relative}")
  endif()
endforeach()
